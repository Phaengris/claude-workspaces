```go
// Package wsp derives everything about a workspace from reality: the
// allocation registry, the filesystem, and git. Nothing here mutates (M1).
package wsp

import (
	"strings"

	"git.internal/cat/claude-workspaces-go/internal/alloc"
	"git.internal/cat/claude-workspaces-go/internal/config"
)

// Subst replaces ${K} for every key in vars. Unknown tokens pass through
// untouched — runtime tokens and load-time params share one syntax, and
// passing unknowns through is what makes that safe (spec §4).
func Subst(s string, vars map[string]string) string {
	for k, v := range vars {
		s = strings.ReplaceAll(s, "${"+k+"}", v)
	}
	return s
}

// RuntimeVars builds the substitution set for one workspace(+project):
// WORKSPACE, PROJECT (only when project != ""), and the index-derived values.
func RuntimeVars(cfg *config.Config, taskID, project string, index int) map[string]string {
	vars := alloc.ComputeValues(cfg.Values, index)
	vars["WORKSPACE"] = taskID
	if project != "" {
		vars["PROJECT"] = project
	}
	return vars
}

// ResolvedEnv merges global env with the project's env (project wins) and
// substitutes runtime tokens in every VALUE. Keys are never substituted.
func ResolvedEnv(cfg *config.Config, taskID, project string, index int) map[string]string {
	vars := RuntimeVars(cfg, taskID, project, index)
	out := make(map[string]string, len(cfg.Env))
	for k, v := range cfg.Env {
		out[k] = Subst(v, vars)
	}
	if p := cfg.Projects[project]; p != nil {
		for k, v := range p.Env {
			out[k] = Subst(v, vars)
		}
	}
	return out
}
```

```go
// Package gitx is read-only git plumbing. Argv form only — no shell ever
// touches a repo path (spec §7).
package gitx

import (
	"bytes"
	"os/exec"
	"strings"
	"sync"
)

func git(dir string, args ...string) (string, error) {
	cmd := exec.Command("git", append([]string{"-C", dir}, args...)...)
	var out bytes.Buffer
	cmd.Stdout = &out
	cmd.Stderr = &out
	err := cmd.Run()
	return strings.TrimSpace(out.String()), err
}

// IsWorkTree reports whether dir is inside a git work tree. Any git failure
// (missing dir, not a repo) is simply "no".
func IsWorkTree(dir string) bool {
	out, err := git(dir, "rev-parse", "--is-inside-work-tree")
	return err == nil && out == "true"
}

// Branch returns the current branch name (or "HEAD" when detached).
func Branch(dir string) (string, error) {
	return git(dir, "rev-parse", "--abbrev-ref", "HEAD")
}

// Stats is one dir's git summary. Err records a per-dir failure so a batch
// never aborts on one bad dir.
type Stats struct {
	Branch string
	Dirty  bool
	Err    error
}

func statsOne(dir string) Stats {
	branch, err := Branch(dir)
	if err != nil {
		return Stats{Err: err}
	}
	porcelain, err := git(dir, "status", "--porcelain")
	if err != nil {
		return Stats{Branch: branch, Err: err}
	}
	return Stats{Branch: branch, Dirty: porcelain != ""}
}

// StatsFor computes Stats for every dir with at most limit concurrent git
// invocations. The semaphore is a buffered channel; the mutex guards the
// shared result map (maps are not safe for concurrent writes).
func StatsFor(dirs []string, limit int) map[string]Stats {
	if limit < 1 {
		limit = 1
	}
	var (
		mu  sync.Mutex
		wg  sync.WaitGroup
		sem = make(chan struct{}, limit)
		out = make(map[string]Stats, len(dirs))
	)
	for _, dir := range dirs {
		wg.Add(1)
		go func() {
			defer wg.Done()
			sem <- struct{}{}
			defer func() { <-sem }()
			s := statsOne(dir)
			mu.Lock()
			out[dir] = s
			mu.Unlock()
		}()
	}
	wg.Wait()
	return out
}
```

(Go ≥1.22 loop variables are per-iteration, so the closure over `dir` is safe — state this in a comment; it's a classic pre-1.22 bug site and worth the teaching note.)
